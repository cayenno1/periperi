rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // ----- Shared validation helpers -----
    function isValidRole(role) {
      return role in ['Staff', 'Admin', 'Manager', 'Owner', 'System_Admin'];
    }
    
    function isValidStatus(status) {
      return status in ['active', 'suspended', 'inactive', 'deleted'];
    }

    // Removed: isValidUnitType, isValidBaseUnit, isValidStockDocument, isValidStockUpdate
    // These were only used for inventory validation which is no longer needed in recipe-based system

    function isNonNegativeNumber(value) {
      return value is number && value >= 0;
    }

    function hasValidTimestamps(data) {
      return (!('createdAt' in data) || data.createdAt is timestamp) &&
             (!('updatedAt' in data) || data.updatedAt is timestamp);
    }

    // ----- Order Counters (for sequential order numbering) -----
    match /order_counters/{counterId} {
      // Allow read and write for order counter generation
      // This is needed for the sequential order ID system (POS)
      allow read, write: if true;
    }

    // ----- Customers -----
    match /customers/{userId} {
      allow create: if request.auth != null && request.auth.uid == userId;
      // Allow read for customers themselves and for admin dashboard (open read for now)
      // TODO: Implement Firebase Auth for staff to restrict this properly
      allow read: if true;
      allow update: if request.auth != null && request.auth.uid == userId;
      allow delete: if request.auth != null && request.auth.uid == userId;

      // ----- Cart Items (subcollection under customers) -----
      match /cartItems/{cartItemId} {
        // Only the customer themselves can manage their cart items
        allow read: if request.auth != null && request.auth.uid == userId;
        allow create: if request.auth != null && 
                         request.auth.uid == userId &&
                         request.resource.data.keys().hasAll(['itemId', 'name', 'price', 'quantity']) &&
                         request.resource.data.price is number &&
                         request.resource.data.quantity is number &&
                         request.resource.data.quantity > 0;
        allow update: if request.auth != null && 
                         request.auth.uid == userId &&
                         request.resource.data.price is number &&
                         request.resource.data.quantity is number &&
                         request.resource.data.quantity > 0;
        allow delete: if request.auth != null && request.auth.uid == userId;
      }
    }

    // ----- Staff -----
    match /staff/{staffId} {
      allow create: if true;
      allow read: if true;

      allow update: if (
        request.resource.data.staffId == resource.data.staffId &&
        (!request.resource.data.diff(resource.data).affectedKeys().hasAny(['password']) ||
         request.resource.data.password == resource.data.password) &&
        (!request.resource.data.diff(resource.data).affectedKeys().hasAny(['firstName']) || 
         (request.resource.data.firstName is string && request.resource.data.firstName.size() > 0)) &&
        (!request.resource.data.diff(resource.data).affectedKeys().hasAny(['lastName']) || 
         (request.resource.data.lastName is string && request.resource.data.lastName.size() > 0)) &&
        (!('email' in request.resource.data.diff(resource.data).affectedKeys()) || 
         (request.resource.data.email is string && request.resource.data.email.matches('.*@.*\\..*'))) &&
        (!('role' in request.resource.data.diff(resource.data).affectedKeys()) || 
         (request.resource.data.role is string && isValidRole(request.resource.data.role))) &&
        (!('status' in request.resource.data.diff(resource.data).affectedKeys()) || 
         (request.resource.data.status is string && isValidStatus(request.resource.data.status)))
      );

      allow delete: if false;
    }

    // ----- Stocks (Inventory) - REMOVED: System is now recipe-based -----
    // Stocks collection is no longer used, but kept commented for reference
    // match /stocks/{stockId} {
    //   allow read: if true;
    //   allow create: if isValidStockDocument(request.resource.data);
    //   allow update: if isValidStockUpdate(request.resource.data, resource.data);
    //   allow delete: if false;
    // }

    // ----- Menu (Dishes) -----
    function isValidMenuItem(data) {
      return data.keys().hasAll(['name', 'category', 'price']) &&
             data.name is string && data.name.size() > 0 &&
             data.category is string && data.category.size() > 0 &&
             isNonNegativeNumber(data.price) &&
             (!('description' in data) || data.description is string) &&
             (!('imageDataUrl' in data) || data.imageDataUrl is string || data.imageDataUrl == null) &&
             (!('ingredients' in data) || data.ingredients is list) &&
             (!('menuId' in data) || data.menuId is string) &&
             (!('slug' in data) || data.slug is string) &&
             hasValidTimestamps(data);
    }

    function isValidMenuItemUpdate(newData, existingData) {
      return isValidMenuItem(newData);
    }

    match /menu/{menuId} {
      allow read: if true;
      allow create: if isValidMenuItem(request.resource.data);
      allow update: if isValidMenuItemUpdate(request.resource.data, resource.data);
      allow delete: if true; // Allow deletes to move items to menuArchives
    }

    // ----- Review validation helper -----
    function isValidReview(data) {
      return data.keys().hasAll(['userId', 'rating']) &&
             data.userId is string &&
             data.userId.size() > 0 &&
             data.rating is number &&
             data.rating >= 1 &&
             data.rating <= 5 &&
             (!('text' in data) || data.text is string) &&
             (!('anonymous' in data) || data.anonymous is bool) &&
             (!('displayName' in data) || data.displayName is string || data.displayName == null) &&
             (!('itemId' in data) || data.itemId is string) &&
             (!('itemName' in data) || data.itemName is string || data.itemName == null) &&
             hasValidTimestamps(data);
    }

    // ----- Reviews (subcollection under menu items) -----
    match /menu/{menuId}/reviews/{reviewId} {
      allow read: if true;

      function isReviewOwner() {
        return request.auth != null && resource.data.userId == request.auth.uid;
      }

      function isReviewOwnerOnCreate() {
        return request.auth != null && request.resource.data.userId == request.auth.uid;
      }

      allow create: if isReviewOwnerOnCreate() && isValidReview(request.resource.data);

      allow update: if isReviewOwner() &&
                    isValidReview(request.resource.data) &&
                    request.resource.data.userId == resource.data.userId &&
                    request.resource.data.createdAt == resource.data.createdAt;

      allow delete: if isReviewOwner();
    }

    // ----- Menu Archives -----
    match /menuArchives/{menuId} {
      allow read: if true;
      allow create: if true; // Allow archiving when menu status becomes inactive or deleted
      allow update: if true; // Allow updates to archived items (for restoring, etc.)
      allow delete: if true; // Allow deletion for cleanup of expired items (30+ days old) - only documents, not collection
    }

    // ----- For Delivery -----
    match /for_delivery/{deliveryId} {
      allow read: if true;
      allow create: if true;
      allow update: if true;
      allow delete: if false;
    }

    // ----- Drivers -----
    match /drivers/{driverId} {
      // Allow reading driver status/availability for admin dashboard
      allow read: if true;
      // Allow creating driver records
      allow create: if true;
      // Allow drivers to update their own status/availability when logging in/out
      allow update: if true;
      // Prevent deletion of driver records
      allow delete: if false;
    }

    // ----- Orders -----
    match /orders/{orderId} {
      // Helper function to validate order status update
      function isValidOrderStatus(status) {
        return status in ['pending', 'new', 'accepted', 'being-cooked', 'cooking', 'preparing', 'ready', 'ready for pick-up', 'ready_for_pickup', 'ready for pickup', 'out_for_delivery', 'out for delivery', 'out-for-delivery', 'delivered', 'cancelled', 'completed', 'canceled'];
      }

      // Helper function to validate order update
      // Note: Since the system uses custom session-based auth (not Firebase Auth),
      // we allow updates with proper validation. Consider implementing Firebase Auth
      // for staff members for better security.
      function isValidOrderUpdate() {
        // Allow status updates
        let statusUpdate = !('status' in request.resource.data.diff(resource.data).affectedKeys()) ||
                          (request.resource.data.status is string && 
                           isValidOrderStatus(request.resource.data.status));
        
        // Allow driver assignment
        let driverUpdate = !('driverId' in request.resource.data.diff(resource.data).affectedKeys()) ||
                          (request.resource.data.driverId is string || 
                           request.resource.data.driverId == null);
        
        // Allow timestamp updates (acceptedAt, updatedAt, cookingStartedAt, assignedAt, readyAt, readyForPickupAt, outForDeliveryAt, deliveredAt)
        let timestampUpdate = (!('acceptedAt' in request.resource.data.diff(resource.data).affectedKeys()) ||
                               request.resource.data.acceptedAt is timestamp) &&
                              (!('updatedAt' in request.resource.data.diff(resource.data).affectedKeys()) ||
                               request.resource.data.updatedAt is timestamp) &&
                              (!('cookingStartedAt' in request.resource.data.diff(resource.data).affectedKeys()) ||
                               request.resource.data.cookingStartedAt is timestamp) &&
                              (!('assignedAt' in request.resource.data.diff(resource.data).affectedKeys()) ||
                               request.resource.data.assignedAt is timestamp) &&
                              (!('readyAt' in request.resource.data.diff(resource.data).affectedKeys()) ||
                               request.resource.data.readyAt is timestamp) &&
                              (!('readyForPickupAt' in request.resource.data.diff(resource.data).affectedKeys()) ||
                               request.resource.data.readyForPickupAt is timestamp) &&
                              (!('outForDeliveryAt' in request.resource.data.diff(resource.data).affectedKeys()) ||
                               request.resource.data.outForDeliveryAt is timestamp) &&
                              (!('deliveredAt' in request.resource.data.diff(resource.data).affectedKeys()) ||
                               request.resource.data.deliveredAt is timestamp);
        
        // Prevent changes to critical fields
        // For guest/walk-in orders, userId might be null, so handle that case
        let userIdUnchanged = 
          (!('userId' in request.resource.data) && !('userId' in resource.data)) ||
          (request.resource.data.userId == resource.data.userId);
        
        let criticalFieldsUnchanged = 
          userIdUnchanged &&
          request.resource.data.total == resource.data.total &&
          request.resource.data.items == resource.data.items;
        
        return statusUpdate && driverUpdate && timestampUpdate && criticalFieldsUnchanged;
      }

      // Customers can create their own orders (requires Firebase Auth)
      // Also allow creation for guest orders and POS/walk-in orders (no auth required)
      function isAuthedOrder() {
        return request.auth != null && request.resource.data.userId == request.auth.uid;
      }

      function isGuestOrder() {
        return request.auth == null &&
               (!('userId' in request.resource.data) || request.resource.data.userId == null) &&
               request.resource.data.isGuest == true;
      }

      // POS/Walk-in orders: allow creation without auth if it's a walk-in order
      function isWalkInOrder() {
        return request.auth == null &&
               request.resource.data.walkin == true &&
               request.resource.data.isGuest == true &&
               request.resource.data.status == 'completed';
      }

      allow create: if isAuthedOrder() || isGuestOrder() || isWalkInOrder();

      // Read access is open so the admin dashboard can list orders
      allow read: if true;

      // Allow updates with proper validation
      // TODO: Implement Firebase Auth for staff to make this more secure
      allow update: if isValidOrderUpdate();
      
      // Deletes should be handled through privileged flows (e.g., Cloud Functions)
      allow delete: if false;
    }

    // ----- Staff Logs (Shift Tracking) -----
    match /logsStaff/{logId} {
      // Helper function to validate shift log document
      function isValidShiftLog(data) {
        return data.keys().hasAll(['driverId', 'driverName', 'startTime']) &&
               data.driverId is string &&
               data.driverId.size() > 0 &&
               data.driverName is string &&
               data.driverName.size() > 0 &&
               data.startTime is timestamp &&
               (!('endTime' in data) || data.endTime is timestamp || data.endTime == null) &&
               (!('createdAt' in data) || data.createdAt is timestamp) &&
               (!('updatedAt' in data) || data.updatedAt is timestamp);
      }

      // Allow drivers to create their own shift logs
      // Note: Since the system uses custom session-based auth, we allow creation with validation
      allow create: if isValidShiftLog(request.resource.data);

      // Allow reading shift logs (drivers can read their own, admins can read all)
      // Open read for now since custom auth is used
      allow read: if true;

      // Allow updating shift logs to add endTime (only if endTime is being added/updated)
      allow update: if (
        // Prevent changing driverId
        request.resource.data.driverId == resource.data.driverId &&
        // Prevent changing driverName
        request.resource.data.driverName == resource.data.driverName &&
        // Prevent changing startTime
        request.resource.data.startTime == resource.data.startTime &&
        // Only allow updating endTime, createdAt, or updatedAt
        (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['endTime', 'updatedAt']) ||
         (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['endTime', 'createdAt', 'updatedAt']))) &&
        // Validate endTime is timestamp or null
        (!('endTime' in request.resource.data) || 
         request.resource.data.endTime is timestamp || 
         request.resource.data.endTime == null) &&
        // Validate updatedAt is timestamp
        (!('updatedAt' in request.resource.data) || 
         request.resource.data.updatedAt is timestamp)
      );

      // Prevent deletion of shift logs (for audit trail)
      allow delete: if false;
    }

    // ----- Promotions -----
    match /promotionList/{promoId} {
      // Helper function to validate promotion document
      function isValidPromotion(data) {
        return data.keys().hasAll(['title', 'description', 'imageUrl', 'startDate', 'endDate', 'placement']) &&
               data.title is string &&
               data.title.size() > 0 &&
               data.description is string &&
               data.description.size() > 0 &&
               data.imageUrl is string &&
               data.imageUrl.size() > 0 &&
               data.startDate is timestamp &&
               data.endDate is timestamp &&
               data.placement is string &&
               data.placement.size() > 0 &&
               (!('createdAt' in data) || data.createdAt is timestamp) &&
               (!('updatedAt' in data) || data.updatedAt is timestamp);
      }

      // Helper function to validate promotion update
      function isValidPromotionUpdate(newData, existingData) {
        return isValidPromotion(newData) &&
               // Ensure startDate and endDate are valid timestamps
               newData.startDate is timestamp &&
               newData.endDate is timestamp &&
               // End date must be after start date
               newData.endDate > newData.startDate;
      }

      // Allow reading promotions (for customer-facing display)
      allow read: if true;

      // Allow creating promotions with validation
      allow create: if isValidPromotion(request.resource.data);

      // Allow updating promotions with validation
      allow update: if isValidPromotionUpdate(request.resource.data, resource.data);

      // Allow deletion for managing promotions
      allow delete: if true;
    }

    // ----- Settings (Stock Status, App Config) -----
    match /settings/{settingId} {
      // Allow reading settings (customer-side needs stock status)
      allow read: if true;
      
      // Allow creating/updating settings (admin dashboard updates stock status)
      allow create: if true;
      allow update: if true;
      
      // Prevent deletion of settings
      allow delete: if false;
    }

    // ----- Table Numbers -----
    match /TableNumber/{tableId} {
      // Helper function to validate table number document
      function isValidTableNumber(data) {
        return data.keys().hasAll(['name', 'code']) &&
               data.name is string &&
               data.name.size() > 0 &&
               data.code is string &&
               data.code.size() > 0 &&
               (!('description' in data) || data.description is string || data.description == null) &&
               (!('isActive' in data) || data.isActive is bool) &&
               (!('createdAt' in data) || data.createdAt is timestamp) &&
               (!('updatedAt' in data) || data.updatedAt is timestamp);
      }

      // Helper function to validate table number update
      function isValidTableNumberUpdate(newData, existingData) {
        return newData.keys().hasAll(['name', 'code']) &&
               newData.name is string &&
               newData.name.size() > 0 &&
               newData.code is string &&
               newData.code.size() > 0 &&
               (!('description' in newData) || newData.description is string || newData.description == null) &&
               (!('isActive' in newData) || newData.isActive is bool) &&
               (!('updatedAt' in newData) || newData.updatedAt is timestamp) &&
               // For updates, code must remain the same (unique identifier), but name can change
               newData.code == existingData.code &&
               // createdAt should not be changed in updates (if present, it must match)
               (!('createdAt' in newData) || newData.createdAt == existingData.createdAt);
      }

      // Allow reading table numbers
      allow read: if true;

      // Allow creating table numbers with validation
      allow create: if isValidTableNumber(request.resource.data);

      // Allow updating table numbers with validation
      allow update: if isValidTableNumberUpdate(request.resource.data, resource.data);

      // Allow deletion for managing table numbers
      allow delete: if true;
    }

    // ----- Daily Servings (Daily Serving Counts) -----
    match /dailyServings/{servingId} {
      // Allow reading daily serving counts (needed for menu availability checks)
      allow read: if true;
      
      // Allow creating daily serving count documents
      // Format: date_menuId (e.g., "2026-01-17_70552033")
      // Fields: menuItemId, menuItemName, date, count, maxServings, createdAt, updatedAt
      allow create: if (
        request.resource.data.count is number &&
        request.resource.data.count >= 0 &&
        request.resource.data.menuItemId is string &&
        request.resource.data.date is string &&
        (!('maxServings' in request.resource.data) || 
         request.resource.data.maxServings == null ||
         (request.resource.data.maxServings is number && request.resource.data.maxServings >= 0))
      );
      
      // Allow updating daily serving counts (incrementing when orders are placed)
      // Only allow incrementing count (not decreasing)
      allow update: if (
        request.resource.data.count is number &&
        request.resource.data.count >= 0 &&
        request.resource.data.count >= resource.data.count &&
        // Prevent changing menuItemId or date
        request.resource.data.menuItemId == resource.data.menuItemId &&
        request.resource.data.date == resource.data.date
      );
      
      // Prevent deletion (for audit trail)
      allow delete: if false;
    }

    // ----- Ingredient Logs (Usage and Restock Tracking) -----
    match /ingredientLogs/{logId} {
      // Helper function to validate ingredient log document
      function isValidIngredientLog(data) {
        return data.keys().hasAll(['ingredientId', 'ingredientName', 'type', 'amount', 'timestamp']) &&
               data.ingredientId is string &&
               data.ingredientId.size() > 0 &&
               data.ingredientName is string &&
               data.ingredientName.size() > 0 &&
               data.type is string &&
               data.type in ['used', 'received'] &&
               data.amount is number &&
               data.amount > 0 &&
               data.timestamp is timestamp &&
               (!('orderId' in data) || data.orderId is string || data.orderId == null) &&
               (!('menuItemName' in data) || data.menuItemName is string || data.menuItemName == null) &&
               (!('date' in data) || data.date is string);
      }

      // Allow reading ingredient logs (for reporting and analytics)
      allow read: if true;
      
      // Allow creating ingredient logs (system logs usage and restocks)
      allow create: if isValidIngredientLog(request.resource.data);
      
      // Prevent updates (logs should be immutable for audit trail)
      allow update: if false;
      
      // Prevent deletion (logs should be kept for audit trail)
      allow delete: if false;
    }
  }
}
